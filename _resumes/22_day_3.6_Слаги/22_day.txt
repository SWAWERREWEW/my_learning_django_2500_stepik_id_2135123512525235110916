import re

def cyrillic_to_latin(text: str) -> str:
    # Словарь соответствия русских букв английским
    rules = {
        'а': 'a', 'б': 'b', 'в': 'v', 'г': 'g', 'д': 'd', 'е': 'e', 'ё': 'yo',
        'ж': 'zh', 'з': 'z', 'и': 'i', 'й': 'y', 'к': 'k', 'л': 'l', 'м': 'm',
        'н': 'n', 'о': 'o', 'п': 'p', 'р': 'r', 'с': 's', 'т': 't', 'у': 'u',
        'ф': 'f', 'х': 'kh', 'ц': 'ts', 'ч': 'ch', 'ш': 'sh', 'щ': 'shch',
        'ы': 'y', 'э': 'e', 'ю': 'yu', 'я': 'ya',
        # Дополнительно переводим пробелы и дефисы
        ' ': '-', '_': '-'
    }

    # Переводим текст в нижний регистр для упрощения
    text = text.lower().strip()

    # Посимвольно заменяем русские буквы на английские
    output = []
    for char in text:
        output.append(rules.get(char, char))  # Если символа нет в словаре, оставляем как есть

    result = "".join(output)

    # Удаляем любые символы, которые НЕ являются буквами, цифрами или дефисом
    result = re.sub(r'[^a-z0-9-]', '', result)

    # Убираем двойные дефисы, если они случайно образовались (например, "--")
    result = re.sub(r'-+', '-', result)

    return result.strip('-')

В учебных целях можно пропустить
Можно сделать автоматическую генерацию слага на основе других заполненных полей дописав (фактически переопределив)
функцию save в модуль Women из файла sitewomen\women\models.py
...
class Women(models.Model):
    ...
    def save(self, *args, **kwargs):
        # Если слаг еще не заполнен (например, при создании нового поста)
        if not self.slug:
            # Переводим название в латиницу через вашу функцию
            latin_title = cyrillic_to_latin(self.title)

            # Генерируем уникальный слаг. Сначала сохраняем объект, чтобы получить pk,
            # либо используем временный слаг, если это создание новой записи:
            self.slug = f"slug-{latin_title}"

        # Вызываем оригинальный метод сохранения Django
        super().save(*args, **kwargs)

        # Если это было первое сохранение и нам нужно вшить туда PK для 100% уникальности:
        # (так как до super().save() у новой записи еще нет PK в базе данных)
        if f"slug-{cyrillic_to_latin(self.title)}" == self.slug:
            clean_date = self.time_update.strftime('%Y-%m-%d-time%H-%M-%S')
            self.slug = f"slug-{self.pk}-{cyrillic_to_latin(self.title)}-date-{clean_date}"
            # Сохраняем еще раз уже с готовым PK и датой
            super().save(update_fields=['slug'])
