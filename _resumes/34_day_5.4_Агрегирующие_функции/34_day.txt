from django.db.models import Count, Sum, Avg, Max, Min
from women.models import Husband

Агрегирующие функции и их применение
Husband.objects.aggregate(Min("age"))
Husband.objects.aggregate(Min("age"), Max("age"))
Husband.objects.aggregate(big=Min("age"), little=Max("age"))
Husband.objects.aggregate(super=Min("age") - Max("age"))
Husband.objects.aggregate(middle_value=Avg("age"))

Метод valies отображает выбранные поля
Women.objects.values("title", "cat_id")
Women.objects.values("title", "cat__name")
Women.objects.values("title", "cat__name").get(id=2)
