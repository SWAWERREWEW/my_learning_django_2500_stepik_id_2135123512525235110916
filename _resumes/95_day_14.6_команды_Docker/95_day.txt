# Просмотр контейнеров
## -a = Все (даже неактивные)
## -q = только идентификаторы
## -f = фильтр с условиями
### -f status=exited завершённые
### -f status=running работающие
### -f status=paused поставленные на паузу через программу
### -f status=created новорождённые контейнеры, но не были ни разу запущенны
### -f status=restarting перезапущенные
### -f exited=130 с кодом 130
docker ps -a

docker run --rm -d --name plot first_image:0.2

Остановка всех работающих контейнеров в powershell
docker stop $(docker ps -qf status=running)

Заморозка или временная остановка работы контейнера
docker pause
Продолжение работы контейнера после паузы или заморозки
docker unpause

Объединение команд. Удаление и остановка контейнеров одновременно
docker rm $(docker stop $(docker ps -q))
