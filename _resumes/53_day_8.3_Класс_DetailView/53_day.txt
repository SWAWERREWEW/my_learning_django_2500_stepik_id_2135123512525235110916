Для отображения деталей поста используется класс DetailView. Им можно заменить функцию show_post.
Параметр slug_url_kwarg позволяет в адресе использовать выбранную переменную

class ShowPost(DetailView):
    context_object_name = 'post'
    template_name = 'women/post.html'
    slug_url_kwargs = 'post_slug'

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['title'] = context['post'].title
        context['menu'] = menu
        return context

    def get_object(self, queryset=None):
        return get_object_or_404(Women.published, slug=self.kwargs[self.slug_url_kwargs])
