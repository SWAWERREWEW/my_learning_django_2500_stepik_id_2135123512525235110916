# sitewomen/users/views.py
from django.contrib.auth import authenticate, login
from django.http import HttpResponse, HttpResponseRedirect
from django.shortcuts import render
from django.urls import reverse
from .forms import LoginUserForm
# Create your views here.
def login_user(request):
    if request.method == 'POST':
        form = LoginUserForm(request.POST)
        if form.is_valid():
            cd = form.cleaned_data
            user = authenticate(request, username=cd["username"], password=cd["password"])
            if user and user.is_active:
                login(request, user)
                return HttpResponseRedirect(reverse("home"))
    else:
        form = LoginUserForm()
    return render(request, 'users/login.html', {'form': form})
def logout_user(request):
    return HttpResponse("logout🛠📤🗝")

После запуска при правильных введённых данных будет перенаправление на главную страницу после формы авторизации по
адресу. После выхода из суперпользователя через админ панель, можно перейти по этому адресу и авторизоваться снова.
http://127.0.0.1:8000/users/login/
